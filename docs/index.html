اتصال کامل به API صرافی نوبیتکس

یک SDK استاندارد و مدرن برای ترید خودکار در صرافی نوبیتکس: مدیریت سفارش‌ها (لیمیت/مارکت/حد ضرر/OCO)، کیف پول و واریز/برداشت، آمار بازار، داده‌های لحظه‌ای از طریق وب‌سوکت Centrifugo، احراز هویت با توکن یا کلید API همراه با امضای Ed25519، مدیریت محدودیت نرخ و تلاش مجدد هوشمند. همهٔ متدها کامنت‌گذاری شده (XML Doc) تا روی GitHub برای همه قابل فهم باشد.

1using System;
2using System.Threading.Tasks;
3using Nobitex.Net;
4using Nobitex.Net.Models;
5 
6// ---------------------------------------------------------------------------
7// Minimal end-to-end sample: read the market, place a limit order, poll it.
8// ---------------------------------------------------------------------------
9 
10var options = new NobitexClientOptions
11{
12    ApiToken = Environment.GetEnvironmentVariable("NOBITEX_TOKEN"),  // از پنل نوبیتکس
13    UserAgent = "TraderBot/MyFirstBot",
14    EnableClientRateLimiting = true
15};
16 
17await using var client = new NobitexClient(options);
18 
19// 1) Market statistics (public - no authentication required)
20var stats = await client.MarketData.GetMarketStatsAsync(srcCurrency: "btc", dstCurrency: "rls");
21var btc = stats.Stats["btc-rls"];
22Console.WriteLine($"BTC best ask: {btc.BestSell:N0} | best bid: {btc.BestBuy:N0} | 24h: {btc.DayChange}%");
23 
24// 2) Order book (public)
25var book = await client.MarketData.GetOrderBookAsync("BTCIRT", size: 50);
26Console.WriteLine($"Top bid: {book.Bids[0].Price} | Top ask: {book.Asks[0].Price}");
27 
28// 3) Wallet balance
29var balance = await client.Wallet.GetBalanceAsync("rls");
30Console.WriteLine($"IRR balance: {balance:N0}");
31 
32// 4) Place a limit buy order
33var request = new PlaceOrderRequest
34{
35    Side         = OrderSide.Buy,
36    Execution    = OrderExecution.Limit,
37    SrcCurrency  = "btc",
38    DstCurrency  = "rls",
39    Amount       = 0.0005m,
40    Price        = 4_400_000_000m,
41    ClientOrderId = $"bot-{DateTimeOffset.UtcNow.ToUnixTimeMilliseconds()}"
42};
43 
44var order = await client.Trading.PlaceOrderAsync(request);
45Console.WriteLine($"Order #{order.Id} -> {order.Status}, unmatched: {order.UnmatchedAmount}");
46 
47// 5) Poll the order until it is finished (or cancelled)
48while (order.Status is OrderStatus.Active or OrderStatus.New)
49{
50    await Task.Delay(TimeSpan.FromSeconds(3));
51    order = await client.Trading.GetOrderAsync(order.Id);
52    Console.WriteLine($"  matched {order.MatchedAmount}/{order.Amount}");
53}
54 
55// 6) Report
56var fills = await client.Trading.GetMyTradesAsync("btc", "rls");
57foreach (var trade in fills.Trades)
58{
59    Console.WriteLine($"{trade.Timestamp:HH:mm:ss} {trade.Type,-4} {trade.Amount} @ {trade.Price:N0}");
60}
FEATURES

چرا Nobitex.Net؟

طراحی‌شده بر اساس دستورالعمل‌های طراحی کتابخانهٔ مایکروسافت: نام‌گذاری PascalCase، متدهای async با پسوند Async، مدل تغییرناپذیر، تزریق HttpClient، پشتیبانی از CancellationToken و ILogger.

⚡

REST کامل نوبیتکس

پوشش بازار، اردربوک v3، معاملات، سفارش‌ها، کیف پول، واریز/برداشت، پروفایل و معاملات تعهدی.

📡

وب‌سوکت Centrifugo

کلاینت اختصاصی با اتصال مجدد خودکار، پینگ/پنگ و Resubscribe کانال‌ها.

🔐

دو روش احراز هویت

توکن بلندمدت یا کلید API با امضای Ed25519 (BouncyCastle) روی timestamp+method+url+body.

🚦

مدیریت Rate Limit

Token-bucket داخلی، خواندن backOff از پاسخ 429 و Retry هوشمند با backoff نمایی.

🧱

مدل‌های strongly-typed

recordها و enumهای تایپ‌شده با تبدیل خودکار مقادیر پولی به decimal.

💥

خطاهای تایپ‌شده

NobitexApiException / RateLimit / OrderRejected با کد خطای قابل فیلتر کردن.

🧪

آمادهٔ تست

طراحی بر پایهٔ اینترفیس و HttpClient تزریقی؛ نمونهٔ xUnit در پوشهٔ tests.

🧩

سازگار با DI

ثبت در ASP.NET Core با IHttpClientFactory و Options pattern.

🇮🇷

مستند فارسی + انگلیسی

XML Doc انگلیسی روی همهٔ متدها و مستندات فارسی روی سایت.

⚠️ نکات مهم بازار

  • قیمت در بازارهای ریالی به ریال است (نه تومان).
  • مقادیر پولی را به‌صورت decimal بفرستید (کتابخانه به‌صورت string سریالایز می‌کند).
  • محدودیت مشترک سفارش‌گذاری: ۳۰۰ درخواست در ۱۰ دقیقه.
  • حداقل ارزش معامله: ۳ میلیون ریال یا ۱۱ تتر.

🛡️ ایمنی

  • کلید خصوصی را فقط در Secret Manager ذخیره کنید.
  • برای کلید API حتماً IP Whitelist تعیین کنید.
  • از clientOrderId برای جلوگیری از سفارش تکراری استفاده کنید.
  • User-Agent پیشنهادی نوبیتکس: TraderBot/XXXXX

146        {
147            _response = response;

157        protected override Task<HttpResponseMessage> SendAsync(
158            HttpRequestMessage request, CancellationToken cancellationToken)
159        {
160            LastUrl = request.RequestUri?.PathAndQuery;
161            LastAuthorization = request.Headers.TryGetValues("Authorization", out var values)
162                ? string.Join(' ', values)
163                : null;
164 
165            if (request.Content is not null)
166            {
167                LastBody = request.Content.ReadAsStringAsync(cancellationToken).GetAwaiter().GetResult();

169

171            {
172                Content = new StringContent(_response, Encoding.UTF8, "application/json"),

REALTIME

وب‌سوکت نوبیتکس (Centrifugo)

آدرس اتصال: wss://ws.nobitex.ir/connection/websocket — حداکثر ۱۰۰ اتصال همزمان برای هر IP و ۴۵۰ کانال برای هر اتصال. کانال‌های خصوصی (پیشوند private:) نیازمند توکن دریافتی از GET /auth/ws/token هستند. کانال‌های خصوصی با الگوی private:channel#{websocket_auth_param} ساخته می‌شوند. پینگ/پنگ به‌صورت خودکار مدیریت می‌شود (پاسخ در کمتر از ۲۵ ثانیه). 

1// ---------------------------------------------------------------------------
2// Realtime market data over the Nobitex WebSocket (Centrifugo protocol).
3//   endpoint : wss://ws.nobitex.ir/connection/websocket
4//   channels : public:orderbook-BTCIRT, public:trades-BTCIRT, ...
5//   private  : private:orders#{websocket_auth_param}   (requires a connection token)
6// ---------------------------------------------------------------------------
7 
8await using var socket = new NobitexSocketClient(new NobitexClientOptions
9{
10    ApiToken = Environment.GetEnvironmentVariable("NOBITEX_TOKEN")
11});
12 
13socket.Connected     += (s, e) => Console.WriteLine($"connected: {e.ClientId}");
14socket.Disconnected  += (s, e) => Console.WriteLine($"disconnected: {e.Reason}");
15socket.ErrorOccurred += (s, e) => Console.WriteLine($"error: {e.Exception.Message}");
16 
17// Public channels ------------------------------------------------------------
18socket.OrderBookUpdated += (s, e) =>
19{
20    var ob = e.OrderBook;
21    Console.WriteLine($"{e.Symbol}  bid={ob.BestBid}  ask={ob.BestAsk}");
22};
23 
24socket.PublicTrade   += (s, e) => Console.WriteLine($"trade {e.Trade.Side} {e.Trade.Volume} @ {e.Trade.Price}");
25socket.MarketStats   += (s, e) => Console.WriteLine($"{e.Symbol} last={e.Stats.Latest}");
26socket.CandleUpdated += (s, e) => Console.WriteLine($"{e.Symbol} {e.Interval} close={e.Candle.Close}");
27 
28// Private channels (own orders / own trades) ---------------------------------
29socket.PrivateOrderUpdated += (s, e) =>
30    Console.WriteLine($"my order #{e.Order.Id} -> {e.Order.Status} matched={e.Order.MatchedAmount}");
31 
32socket.PrivateTrade += (s, e) => Console.WriteLine($"my fill {e.Trade.Amount} @ {e.Trade.Price}");
33 
34await socket.ConnectAsync();
35 
36await socket.SubscribeOrderBookAsync("BTCIRT");
37await socket.SubscribeOrderBookAsync("USDTIRT");
38await socket.SubscribeTradesAsync("ETHIRT");
39await socket.SubscribeMarketStatsAsync("BTCIRT");
40await socket.SubscribePrivateOrdersAsync();   // uses GET /auth/ws/token automatically
41await socket.SubscribePrivateTradesAsync();
42 
43// Optional: maintain a local order book snapshot automatically
44var localBook = socket.CreateLocalOrderBook("BTCIRT");
45Console.WriteLine(localBook.Current.BestBid);
46 
47// Raw access to any channel (typed callback + optional delta)
48await socket.SubscribeAsync(
49    "public:orderbook-ADAIRT",
50    delta: "fossil",
51    onPublication: data => Console.WriteLine(data.ToString()));
52 
53// Graceful shutdown
54await socket.UnsubscribeAsync("public:orderbook-BTCIRT");
55await socket.DisconnectAsync();

کانال‌ها

  • public:orderbook-{SYMBOL}دفتر معاملات لحظه‌ای (bids/asks)
  • public:trades-{SYMBOL}معاملات انجام‌شده
  • public:candles-{SYMBOL}کندل‌ها
  • public:stats-{SYMBOL}آمار بازار / آخرین قیمت
  • private:orders#{auth}سفارش‌های خود کاربر
  • private:trades#{auth}معاملات خود کاربر

ویژگی‌های کلاینت

  • اتصال مجدد خودکار با backoff نمایی + jitter
  • Resubscribe خودکار کانال‌ها پس از قطعی
  • پشتیبانی از Delta فسیل و Snapshot
  • نگهدارندهٔ LocalOrderBook برای ساخت دفتر معاملاتی محلی
  • رویدادها به‌صورت event و IAsyncEnumerable

11        Amount = 0.0001m, Price = 4_500_000_000m
12    });
13}
14catch (NobitexOrderRejectedException ex) when (ex.Code == NobitexErrorCode.SmallOrder)

16    // Minimum order value is 3,000,000 IRR (or 11 USDT)

18}
19catch (NobitexRateLimitException ex)
20{
21    // The library already retries using ex.BackOff - handle only when retries are exhausted.
22    Console.WriteLine($"Rate limited. Retry after {ex.BackOff.TotalSeconds:N0}s (limit: {ex.Limit})");
23}
24catch (NobitexApiException ex) when (ex.Code == NobitexErrorCode.BadPrice)
25{
26    Console.WriteLine("قیمت سفارش بیش از ۳۰٪ با بازار فاصله دارد.");
27}
28catch (NobitexApiException ex)

31}
